Python

def gray(encode, n):
    if encode:
        return n ^ (n >> 1)
    b = n
    while n:
        n >>= 1
        b ^= n
    return b

public static int gray(boolean encode, int n) {
    if (encode) return n ^ (n >>> 1);
    int b = n;
    while (n != 0) {
        n >>>= 1;
        b ^= n;
    }
    return b;
}
